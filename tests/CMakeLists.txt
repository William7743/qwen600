cmake_minimum_required(VERSION 3.20)
# Standalone CPU build permits a separate ASan compiler without changing nvcc's host compiler.
if (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(QWEN_CPU_CHECKS LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 11)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()
get_filename_component(QWEN_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
include("${QWEN_SOURCE_ROOT}/cmake/TokenizerDependencies.cmake")
option(QWEN_ENABLE_ASAN "Instrument CPU probes with AddressSanitizer" ON)

if (QWEN_ENABLE_ASAN)
    include(CheckCXXSourceCompiles)
    set(CMAKE_REQUIRED_FLAGS "-fsanitize=address")
    check_cxx_source_compiles("int main() { return 0; }" QWEN_ASAN_AVAILABLE)
    unset(CMAKE_REQUIRED_FLAGS)
    if (NOT QWEN_ASAN_AVAILABLE)
        message(FATAL_ERROR "ASan compiler/runtime unavailable. Install it or build tests/ separately with an ASan-capable compiler. Do not label a non-ASan run as a memory check.")
    endif()
endif()

foreach(probe IN ITEMS tokenizer_probe edge_probe)
    add_executable(${probe} "${probe}.cpp")
    target_include_directories(${probe} PRIVATE "${QWEN_SOURCE_ROOT}")
    target_link_libraries(${probe} PRIVATE qwen_tokenizer_dependencies)
    set_target_properties(${probe} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    if (QWEN_ENABLE_ASAN)
        target_compile_options(${probe} PRIVATE -O1 -g -fsanitize=address -fno-omit-frame-pointer)
        target_link_options(${probe} PRIVATE -fsanitize=address)
    endif()
endforeach()

if (CMAKE_CUDA_COMPILER_LOADED)
    add_executable(operator_probe operator_probe.cu)
    target_include_directories(operator_probe PRIVATE "${QWEN_SOURCE_ROOT}")
    target_link_libraries(operator_probe PRIVATE CUDA::cublas)
    target_compile_options(operator_probe PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
    set_target_properties(operator_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    add_executable(correctness_probe correctness_probe.cu)
    target_include_directories(correctness_probe PRIVATE "${QWEN_SOURCE_ROOT}")
    target_link_libraries(correctness_probe PRIVATE CUDA::cublas qwen_tokenizer_dependencies)
    target_compile_options(correctness_probe PRIVATE
        $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>
        $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
    set_target_properties(correctness_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
